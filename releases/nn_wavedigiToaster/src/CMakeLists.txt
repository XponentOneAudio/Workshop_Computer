cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
project(wavedigitoaster C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()

add_executable(wavedigitoaster wavedigitoaster.cpp)
target_compile_options(wavedigitoaster PRIVATE -O2 -Wall -Wdouble-promotion)
target_link_options(wavedigitoaster PRIVATE -Wl,--print-memory-usage)
target_compile_definitions(wavedigitoaster PRIVATE PICO_XOSC_STARTUP_DELAY_MULTIPLIER=64)
target_include_directories(wavedigitoaster PUBLIC ${CMAKE_CURRENT_LIST_DIR})
target_link_libraries(wavedigitoaster pico_unique_id pico_stdlib hardware_dma hardware_i2c hardware_pwm hardware_adc hardware_spi hardware_flash)
pico_enable_stdio_usb(wavedigitoaster 0)
pico_add_extra_outputs(wavedigitoaster)
