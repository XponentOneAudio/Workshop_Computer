cmake_minimum_required(VERSION 3.13)
include(../../src/pico_sdk_import.cmake)
project(toaster_bench C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
add_executable(toaster_bench bench.cpp)
target_compile_options(toaster_bench PRIVATE -O2)
target_include_directories(toaster_bench PUBLIC ${CMAKE_CURRENT_LIST_DIR}/../../src)
target_link_libraries(toaster_bench pico_stdlib)
pico_set_binary_type(toaster_bench copy_to_ram)
pico_enable_stdio_uart(toaster_bench 1)
pico_enable_stdio_usb(toaster_bench 1)
pico_add_extra_outputs(toaster_bench)
